```bash
grep -rEn 'TD-[0-9]+|RISK-[0-9]+|STORE-[0-9]+|FS-[0-9]+|RD-[0-9]+|\(D[0-9]+[a-z]?\)|[Ii]ssue [0-9]{2,3}|PROVISIONAL|pending [Ii]ssue|PR #[0-9]+|linear\.app|\bMatt\b|signs? off|sign-off' \
